import test from 'node:test';import assert from 'node:assert/strict';import fs from 'node:fs';import os from 'node:os';import path from 'node:path';import {fork} from 'node:child_process';import {EmpireDatabase} from '../../brain/sqlite-database.js';
async function probe(file:string){const c=fork(new URL('./fixtures/sqljs-process-owner-child.ts',import.meta.url),[file],{stdio:['ignore','ignore','ignore','ipc']});const exit=new Promise(r=>c.once('exit',r));try{return await new Promise<any>((resolve,reject)=>{const timer=setTimeout(()=>reject(Error('deadline')),8000);c.once('message',m=>{clearTimeout(timer);resolve(m);});});}finally{c.kill('SIGKILL');await exit;}}
test('failed production constructor releases kernel lock for repaired file',async()=>{const root=fs.mkdtempSync(path.join(os.tmpdir(),'sqljs-constructor-')),file=path.join(root,'brain.db'),prior=process.env.NODE_ENV;fs.writeFileSync(file,'malformed');try{process.env.NODE_ENV='production';assert.throws(()=>new EmpireDatabase(file),/open refused/);fs.unlinkSync(file);assert.equal((await probe(file)).kind,'opened');}finally{if(prior===undefined)delete process.env.NODE_ENV;else process.env.NODE_ENV=prior;fs.rmSync(root,{recursive:true,force:true});}});
test('async flush and shutdown failure retain process ownership until successful close',async t=>{const root=fs.mkdtempSync(path.join(os.tmpdir(),'sqljs-flush-fail-')),file=path.join(root,'brain.db');const db=new EmpireDatabase(file);try{db.exec('CREATE TABLE events(id INTEGER)');await db.requestCriticalPersist();db.exec('INSERT INTO events VALUES(1)');const original=fs.renameSync;t.mock.method(fs,'renameSync',((a:any,b:any)=>{if(b===file)throw Error('REVIEW_RENAME_FAILURE');return original(a,b);}) as any);await assert.rejects(db.requestCriticalPersist(),/REVIEW_RENAME_FAILURE/);assert.match((await probe(file)).error,/OWNERSHIP_UNAVAILABLE/);assert.throws(()=>db.close(),/REVIEW_RENAME_FAILURE/);assert.match((await probe(file)).error,/OWNERSHIP_UNAVAILABLE/);t.mock.restoreAll();db.close();assert.equal((await probe(file)).kind,'opened');}finally{t.mock.restoreAll();db.close();fs.rmSync(root,{recursive:true,force:true});}});
